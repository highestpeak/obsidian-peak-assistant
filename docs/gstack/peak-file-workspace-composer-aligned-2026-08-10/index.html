from pathlib import Path
from peak.index import DocumentIndex

async def refresh_sources(vault: Path) -> int:
    """Refresh changed sources and return the indexed count."""
    index = DocumentIndex(vault)
    changed = await index.find_changed()

    if not changed:
        return 0

    for source in changed:
        await index.upsert(source)

    return len(changed)
