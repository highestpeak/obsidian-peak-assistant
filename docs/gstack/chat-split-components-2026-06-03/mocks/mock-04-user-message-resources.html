# Concurrent failover benchmark — runs N attempts in parallel and tracks fail-rate by provider tier
async def benchmark(providers: list[Provider], attempts: int = 1000, concurrency: int = 32) -> FailoverReport:
    sem = asyncio.Semaphore(concurrency)
    async def probe(_):
        async with sem:
            return await call_any_provider(providers, timeout=5.0)
    results = await asyncio.gather(*(probe(i) for i in range(attempts)), return_exceptions=True)
    failed = sum(1 for r in results if isinstance(r, Exception))
    return FailoverReport(attempts=attempts, failed=failed, rate=failed / attempts)
